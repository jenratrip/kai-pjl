    --}}
    @php
        $dinasHeader = auth()->check()
            ? \App\Models\Dinas::where('user_id', auth()->id())
                ->where('status', 'berlangsung')
                ->latest('id')
                ->first()
            : null;

        $jenisPetugas = $dinasHeader?->is_pengganti ? 'wakilan' : 'reguler';
